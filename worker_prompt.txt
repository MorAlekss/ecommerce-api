**retry_request must become async:**
```python
# BEFORE
def retry_request(func, max_retries=3, backoff=1.0):
    last_error = None
    for attempt in range(max_retries):
        try:
            return func()
        except requests.exceptions.RequestException as e:
            last_error = e
            if attempt < max_retries - 1:
                time.sleep(backoff * (2 ** attempt))
    raise last_error

# AFTER
async def retry_request(func, max_retries=3, backoff=1.0):
    last_error = None
    for attempt in range(max_retries):
        try:
            return await func()
        except httpx.RequestError as e:
            last_error = e
            if attempt < max_retries - 1:
                time.sleep(backoff * (2 ** attempt))
    raise last_error
```

**get_with_retry must become async with httpx.AsyncClient:**
```python
# BEFORE
def get_with_retry(url, params=None, headers=None, max_retries=3):
    def _request():
        response = requests.get(url, params=params, headers=headers)
        response.raise_for_status()
        return response.json()
    return retry_request(_request, max_retries=max_retries)

# AFTER
async def get_with_retry(url, params=None, headers=None, max_retries=3):
    async def _request():
        async with httpx.AsyncClient() as client:
            response = await client.get(url, params=params, headers=headers)
            response.raise_for_status()
            return response.json()
    return await retry_request(_request, max_retries=max_retries)
```

**Same pattern for post_with_retry:**
```python
# BEFORE
def post_with_retry(url, data=None, headers=None, max_retries=3):
    def _request():
        response = requests.post(url, json=data, headers=headers)
        response.raise_for_status()
        return response.json()
    return retry_request(_request, max_retries=max_retries)

# AFTER
async def post_with_retry(url, data=None, headers=None, max_retries=3):

            response = await client.post(url, json=data, headers=headers)
            response.raise_for_status()
            return response.json()
    return await retry_request(_request, max_retries=max_retries)
```

**Key rules:**
1. EVERY function that uses requests must become `async def`
2. Inner `_request` closures must also become `async def` and use `await`
3. The `retry_request` call itself must use `await`
4. `except requests.exceptions.RequestException` → `except httpx.RequestError`
